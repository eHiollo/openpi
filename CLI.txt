--------启动 VLA 桥接器 (异步推理, 机器人 PC 上运行)----------
 # 需要 conda 环境 client（含 opencv/websockets/msgpack），或 pip install -r requirements.txt
 conda activate client
 cd $HOME/Allen/openpi/third_party/A10_new/client
 # 异步双缓冲（推荐）：推理与执行重叠，消除 batch 间 GPU 空闲间隙
 python3 run_bridge.py \
  --policy-host <GPU_IP> --policy-port 8000 \
  --robot-host 127.0.0.1 --robot-port 8080 \
  --mode async --camera-mode threaded \
  --hz 10 --start-delay 1.0
 # 对比/回退：
 #   --mode sync              原同步模式
 #   --camera-mode exclusive  多相机同 USB 总线掉帧时的 fallback
 # 也可从仓库根目录运行：python client/run_bridge.py ... 或 python -m client.run_bridge ...
 # 停止：Ctrl+C（自动发 STOP_POLICY 停机器人并关连接）


--------数据集回放验证（5090 → 机器人 192.168.1.12:8080）----------
 # 机器人：git checkout dev/async-inference && git pull（含 chunk 间接缝 retarget）后编 kaanhbin
 # CLI：policy -x=2 -a=0.15 ；姿态接近 ep15 起点
 cd $HOME/Allen/openpi
 # 同步（已测通）：走完一段 → 假推理 ~68ms（臂停着）→ 再发下一段
 uv run python third_party/A10_new/client/run_bridge.py \
  --policy replay --mode sync --no-camera \
  --replay-dataset $HOME/Allen/openpi/dataset/dataset_5_9 \
  --replay-episode 15 \
  --robot-host 192.168.1.12 --robot-port 8080 \
  --robot-timeout 5 --hz 0 --start-delay 1.0
 # 异步（真正 receding horizon）：执行 chunk N 时采 obs → ~68ms 推出 N+1 → 替换未走完队列；驱动从当前指令接到新队首
 uv run python third_party/A10_new/client/run_bridge.py \
  --policy replay --mode async --no-camera \
  --replay-dataset $HOME/Allen/openpi/dataset/dataset_5_9 \
  --replay-episode 15 \
  --robot-host 192.168.1.12 --robot-port 8080 \
  --robot-timeout 5 --hz 0 --start-delay 1.0 \
  --monitor-hz 5
 # 成功判据：从第二段起日志 rem_before>0；机器人打印 retarget mid-segment；段间不应再有 ~68ms queue_empty。
 # 回放正常结束不发 STOP_POLICY。Ctrl+C 才停。
